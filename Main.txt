local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local Workspace = game:GetService("Workspace")

local LocalPlayer = Players.LocalPlayer

local function GetSyncAPI()
    local tool
    for _, v in ipairs(LocalPlayer:GetDescendants()) do
        if v.Name == "SyncAPI" then tool = v.Parent break end
    end
    if not tool then
        for _, v in ipairs(ReplicatedStorage:GetDescendants()) do
            if v.Name == "SyncAPI" then tool = v.Parent break end
        end
    end
    return tool and tool:FindFirstChild("SyncAPI") and tool.SyncAPI:FindFirstChild("ServerEndpoint")
end

local function InvokeF3X(...)
    local remote = GetSyncAPI()
    if remote then
        remote:InvokeServer(...)
    end
end

local F3X = {
    CreatePart = function(cf, parent)
        InvokeF3X("CreatePart", "Normal", cf, parent or Workspace)
    end,
    SetAnchor = function(boolean, part)
        InvokeF3X("SyncAnchor", {{["Part"] = part, ["Anchored"] = boolean}})
    end,
    SetCollision = function(part, boolean)
        InvokeF3X("SyncCollision", {{["Part"] = part, ["CanCollide"] = boolean}})
    end,
    Resize = function(part, size, cf)
        InvokeF3X("SyncResize", {{["Part"] = part, ["CFrame"] = cf, ["Size"] = size}})
    end,
    AddMesh = function(part)
        InvokeF3X("CreateMeshes", {{["Part"] = part}})
    end,
    SetMesh = function(part, meshid)
        InvokeF3X("SyncMesh", {{["Part"] = part, ["MeshId"] = "rbxassetid://" .. meshid}})
    end,
    SetTexture = function(part, texid)
        InvokeF3X("SyncMesh", {{["Part"] = part, ["TextureId"] = "rbxassetid://" .. texid}})
    end,
    MeshResize = function(part, size)
        InvokeF3X("SyncMesh", {{["Part"] = part, ["Scale"] = size}})
    end,
    SetName = function(part, name)
        InvokeF3X("SetName", {{part}}, name)
    end,
    SetLocked = function(part, boolean)
        InvokeF3X("SetLocked", {{part}}, boolean)
    end,
    SpawnDecal = function(part, side)
        InvokeF3X("CreateTextures", {{["Part"] = part, ["Face"] = side, ["TextureType"] = "Decal"}})
    end,
    AddDecal = function(part, asset, side)
        InvokeF3X("SyncTexture", {{["Part"] = part, ["Face"] = side, ["TextureType"] = "Decal", ["Texture"] = "rbxassetid://" .. asset}})
    end,
    DestroyPart = function(part)
        InvokeF3X("Remove", {{part}})
    end
}

local Actions = {}

function Actions.GetF3X()
    local RequestCommand = ReplicatedStorage:WaitForChild("HDAdminHDClient").Signals.RequestCommand
    RequestCommand:InvokeServer(";btools")
    RequestCommand:InvokeServer(";buildingTools")
end

function Actions.SetSkybox(id)
    local char = LocalPlayer.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") then return end
    local pos = char.HumanoidRootPart.CFrame
    local e, f, g = pos.x, pos.y, pos.z
    F3X.CreatePart(CFrame.new(math.floor(e), math.floor(f), math.floor(g)) + Vector3.new(0, 6, 0), Workspace)
    
    for _, v in ipairs(Workspace:GetDescendants()) do
        if v:IsA("BasePart") and v.CFrame.x == math.floor(e) and v.CFrame.z == math.floor(g) then
            F3X.SetName(v, "Sky")
            F3X.AddMesh(v)
            F3X.SetMesh(v, "111891702759441")
            F3X.SetTexture(v, id)
            F3X.MeshResize(v, Vector3.new(3600, 3600, 3600))
            F3X.SetLocked(v, true)
        end
    end
end

function Actions.SpamDecal(id)
    for _, v in ipairs(Workspace:GetDescendants()) do
        if v:IsA("BasePart") then
            task.spawn(function()
                F3X.SetLocked(v, false)
                for _, side in ipairs(Enum.NormalId:GetEnumItems()) do
                    F3X.SpawnDecal(v, side)
                    F3X.AddDecal(v, id, side)
                end
            end)
        end
    end
end

function Actions.FireAll()
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("BasePart") then
            InvokeF3X("CreateDecorations", {{["Part"] = obj, ["DecorationType"] = "Fire"}})
            InvokeF3X("SyncDecorate", {{["Part"] = obj, ["DecorationType"] = "Fire", ["Size"] = 20}})
        end
    end
end

function Actions.DeleteAll()
    for _, p in ipairs(Players:GetPlayers()) do
        if p.Character then
            pcall(function() F3X.DestroyPart(p.Character) end)
        end
    end
    task.wait(0.5)
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("Part") or obj:IsA("MeshPart") then
            F3X.DestroyPart(obj)
        end
    end
end

function Actions.BurgerRain()
    task.spawn(function()
        while true do
            task.wait(0.1)
            if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
                local hrpcf = LocalPlayer.Character.HumanoidRootPart.CFrame
                local xloc = math.random(-600, 600) + hrpcf.x
                local zloc = math.random(-600, 600) + hrpcf.z
                local yloc = hrpcf.y + math.random(400, 800)

                task.spawn(function()
                    F3X.CreatePart(CFrame.new(math.floor(xloc), yloc, math.floor(zloc)))
                    for _, v in ipairs(Workspace:GetDescendants()) do
                        if v.Name == "Part" and v.Parent == Workspace and v.CFrame.x == math.floor(xloc) and v.CFrame.z == math.floor(zloc) then
                            F3X.SetName(v, "GET BURGERED!")
                            F3X.SetAnchor(false, v)
                            F3X.AddMesh(v)
                            F3X.Resize(v, Vector3.new(45, 35, 40), v.CFrame)
                            F3X.MeshResize(v, Vector3.new(45, 35, 40))
                            F3X.SetMesh(v, "16646125")
                            F3X.SetTexture(v, "16432575")
                            F3X.SetCollision(v, true)
                        end
                    end
                end)
            end
        end
    end)
end
